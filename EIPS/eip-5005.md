## 摘要
此 EIP 标准化了可组合且可互操作的以太坊可编程账户工具的接口。这些接口将合约账户（“化身”）与它们的身份验证和执行逻辑（“守卫”和“模块”）分离。化身实现 `IAvatar` 接口，守卫实现 `IGuard` 接口。模块可以采用任何形式。

## 动机
目前，大多数可编程账户（如 DAO 工具和框架）都构建为单片系统，其中授权和执行逻辑耦合在一起，无论是在同一合约中还是在紧密集成的合约系统中。这不必要地抑制了这些工具的灵活性，并通过高转换成本鼓励平台锁定。

通过使用此 EIP 标准来分离关注点（解耦身份验证和执行逻辑），用户可以：

1. 实现对可编程账户的灵活的、基于模块的控制
2. 轻松地在工具和框架之间切换，而无需不必要的开销。
3. 并行启用多种控制机制。
4. 实现跨链/跨层治理。
5. 随着他们的项目和社区的成熟，逐步下放他们的治理权。

## 规范
本文档中使用的关键词“必须”、“禁止”、“需要”、“应该”、“不应该”、“推荐”、“可以”和“可选”应按照 RFC 2119 中的描述进行解释。

此 EIP 由四个关键概念组成：

- **化身 (Avatars)** 是可编程的以太坊账户。化身是持有余额、拥有系统、执行交易、被外部引用并最终代表你的 DAO 的地址。化身必须实现 `IAvatar` 接口。
- **模块 (Modules)** 是由化身启用的合约，用于实现某些执行逻辑。
- **修饰器 (Modifiers)** 是位于模块和化身之间的合约，用于修改模块的行为。例如，它们可能会对模块尝试执行的所有函数强制执行延迟，或者限制可以由模块发起的交易范围。修饰器必须实现 `IAvatar` 接口。
- **守卫 (Guards)** 是可以在模块或修饰器上启用的合约，并且对这些模块或修饰器执行的每个交易执行预检查或后检查。这允许化身执行诸如限制模块或修饰器可以调用的地址和函数的范围，或确保模块或修饰器永远不会更改某些状态之类的操作。守卫必须公开 `IGuard` 接口。希望可以被守卫的模块、修饰器和化身必须继承 `Guardable`，必须在触发目标执行之前调用 `checkTransaction()`，并且必须在执行完成后调用 `checkAfterExecution()`。

```solidity
/// @title Avatar - A contract that manages modules that can execute transactions via this contract.
// @title 化身 - 管理可以通过此合约执行交易的模块的合约。

pragma solidity >=0.7.0 <0.9.0;

import "./Enum.sol";


interface IAvatar {
    event EnabledModule(address module);
    event DisabledModule(address module);
    event ExecutionFromModuleSuccess(address indexed module);
    event ExecutionFromModuleFailure(address indexed module);

    /// @dev Enables a module on the avatar.
    // @dev 在化身上启用一个模块。
    /// @notice Can only be called by the avatar.
    // @notice 只能由化身调用。
    /// @notice Modules should be stored as a linked list.
    // @notice 模块应存储为链表。
    /// @notice Must emit EnabledModule(address module) if successful.
    // @notice 如果成功，必须发出 EnabledModule(address module)。
    /// @param module Module to be enabled.
    // @param module 要启用的模块。
    function enableModule(address module) external;

    /// @dev Disables a module on the avatar.
    // @dev 在化身上禁用一个模块。
    /// @notice Can only be called by the avatar.
    // @notice 只能由化身调用。
    /// @notice Must emit DisabledModule(address module) if successful.
    // @notice 如果成功，必须发出 DisabledModule(address module)。
    /// @param prevModule Address that pointed to the module to be removed in the linked list
    // @param prevModule 指向链表中要删除的模块的地址
    /// @param module Module to be removed.
    // @param module 要删除的模块。
    function disableModule(address prevModule, address module) external;

    /// @dev Allows a Module to execute a transaction.
    // @dev 允许模块执行交易。
    /// @notice Can only be called by an enabled module.
    // @notice 只能由已启用的模块调用。
    /// @notice Must emit ExecutionFromModuleSuccess(address module) if successful.
    // @notice 如果成功，必须发出 ExecutionFromModuleSuccess(address module)。
    /// @notice Must emit ExecutionFromModuleFailure(address module) if unsuccessful.
    // @notice 如果不成功，必须发出 ExecutionFromModuleFailure(address module)。
    /// @param to Destination address of module transaction.
    // @param to 模块交易的目标地址。
    /// @param value Ether value of module transaction.
    // @param value 模块交易的以太币值。
    /// @param data Data payload of module transaction.
    // @param data 模块交易的数据负载。
    /// @param operation Operation type of module transaction: 0 == call, 1 == delegate call.
    // @param operation 模块交易的操作类型：0 == call，1 == delegate call。
    function execTransactionFromModule(
        address to,
        uint256 value,
        bytes memory data,
        Enum.Operation operation
    ) external returns (bool success);

    /// @dev Allows a Module to execute a transaction and return data
    // @dev 允许模块执行交易并返回数据
    /// @notice Can only be called by an enabled module.
    // @notice 只能由已启用的模块调用。
    /// @notice Must emit ExecutionFromModuleSuccess(address module) if successful.
    // @notice 如果成功，必须发出 ExecutionFromModuleSuccess(address module)。
    /// @notice Must emit ExecutionFromModuleFailure(address module) if unsuccessful.
    // @notice 如果不成功，必须发出 ExecutionFromModuleFailure(address module)。
    /// @param to Destination address of module transaction.
    // @param to 模块交易的目标地址。
    /// @param value Ether value of module transaction.
    // @param value 模块交易的以太币值。
    /// @param data Data payload of module transaction.
    // @param data 模块交易的数据负载。
    /// @param operation Operation type of module transaction: 0 == call, 1 == delegate call。
    // @param operation 模块交易的操作类型：0 == call，1 == delegate call。
    function execTransactionFromModuleReturnData(
        address to,
        uint256 value,
        bytes memory data,
        Enum.Operation operation
    ) external returns (bool success, bytes memory returnData);

    /// @dev Returns if an module is enabled
    // @dev 返回一个模块是否启用
    /// @return True if the module is enabled
    // @return 如果模块已启用，则为 True
    function isModuleEnabled(address module) external view returns (bool);

    /// @dev Returns array of modules.
    // @dev 返回模块数组。
    /// @param start Start of the page.
    // @param start 页面的开始。
    /// @param pageSize Maximum number of modules that should be returned.
    // @param pageSize 应该返回的最大模块数量。
    /// @return array Array of modules.
    // @return array 模块数组。
    /// @return next Start of the next page.
    // @return next 下一页的开始。
    function getModulesPaginated(address start, uint256 pageSize)
        external
        view
        returns (address[] memory array, address next);
}
```

```solidity
pragma solidity >=0.7.0 <0.9.0;

import "./Enum.sol";
import "./BaseGuard.sol";

/// @title Guardable - A contract that manages fallback calls made to this contract
// @title Guardable - 管理对此合约进行的后备调用的合约
contract Guardable {
    address public guard;

    event ChangedGuard(address guard);

    /// `guard_` does not implement IERC165.
    // `guard_` 未实现 IERC165。
    error NotIERC165Compliant(address guard_);

    /// @dev Set a guard that checks transactions before execution.
    // @dev 设置一个在交易执行之前检查交易的守卫。
    /// @param _guard The address of the guard to be used or the 0 address to disable the guard.
    // @param _guard 要使用的守卫的地址，或者用于禁用守卫的 0 地址。
    function setGuard(address _guard) external {
        if (_guard != address(0)) {
            if (!BaseGuard(_guard).supportsInterface(type(IGuard).interfaceId))
                revert NotIERC165Compliant(_guard);
        }
        guard = _guard;
        emit ChangedGuard(guard);
    }

    function getGuard() external view returns (address _guard) {
        return guard;
    }
}
```

```solidity
pragma solidity >=0.7.0 <0.9.0;

import "./Enum.sol";
import "./IERC165.sol";
import "./IGuard.sol";

abstract contract BaseGuard is IERC165 {
    function supportsInterface(bytes4 interfaceId)
        external
        pure
        override
        returns (bool)
    {
        return
            interfaceId == type(IGuard).interfaceId || // 0xe6d7a83a
            interfaceId == type(IERC165).interfaceId; // 0x01ffc9a7
    }

    /// @dev Module transactions only use the first four parameters: to, value, data, and operation.
    // @dev 模块交易仅使用前四个参数：to、value、data 和 operation。
    /// Module.sol hardcodes the remaining parameters as 0 since they are not used for module transactions.
    // Module.sol 将剩余参数硬编码为 0，因为它们不用于模块交易。
    function checkTransaction(
        address to,
        uint256 value,
        bytes memory data,
        Enum.Operation operation,
        uint256 safeTxGas,
        uint256 baseGas,
        uint256 gasPrice,
        address gasToken,
        address payable refundReceiver,
        bytes memory signatures,
        address msgSender
    ) external virtual;

    function checkAfterExecution(bytes32 txHash, bool success) external virtual;
}
```

## 理由
此标准中定义的接口旨在与目前使用的大多数流行的可编程账户最大程度地兼容，以最大限度地减少对现有工具进行更改的需要。

## 向后兼容性
此标准未引入任何向后兼容性问题。

## 安全考虑
模块开发者和用户应考虑以下几点：
1. **模块拥有绝对控制权：** 模块对其启用的任何化身都具有绝对控制权，因此任何模块实现都应被视为安全关键，用户应非常谨慎地启用新模块。仅启用您信任的模块，使其具有化身的全部价值。
2. **竞争条件：** 给定的化身可以启用任意数量的模块，每个模块都单方面控制着 safe。在这种情况下，不同的模块和/或其他控制机制之间可能存在竞争条件。
3. **不要破坏你的化身：** 没有安全措施可以阻止你添加或删除模块。如果您删除所有允许您控制化身的模块，则该化身将停止运行，并且所有资金将被卡住。
